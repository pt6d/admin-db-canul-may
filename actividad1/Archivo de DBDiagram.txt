Table usuarios {
  id_usuario integer [primary key]
  nombre_usuario varchar
  email_usuario varchar
}

Table libros {
  id_libro integer [primary key]
  titulo_libro varchar
  autor_libro varchar
}

Table prestamos {
  id_prestamo integer [primary key]
  id_usuario integer [not null]
  id_libro integer [not null]
  fecha_prestamo date
}

Ref usuario_prestamos: prestamos.id_usuario > usuarios.id_usuario
Ref libro_prestamos: prestamos.id_libro > libros.id_libro
